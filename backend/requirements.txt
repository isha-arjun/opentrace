from __future__ import annotations

import pytest

from backend.app.observatory import build_scan_report


@pytest.fixture
def sample_payload():
    return {
        "scan_id": "abc123",
        "state": "FINISHED",
        "tests": {
            "content-security-policy": {"score": 90, "status": "pass", "observed": "default-src 'self'"},
            "strict-transport-security": {"score": 50, "status": "warning", "observed": "max-age=31536000"},
            "referrer-policy": {"score": 0, "status": "fail", "observed": "strict-origin-when-cross-origin"},
            "x-content-type-options": {"score": 100, "status": "pass", "observed": "nosniff"},
            "x-frame-options": {"score": 90, "status": "pass", "observed": "DENY"},
        },
    }


def test_build_scan_report_parses_supported_findings(sample_payload):
    report = build_scan_report("https://example.com/", sample_payload)
    assert report.summary.total == 5
    assert report.findings[0].title == "Content-Security-Policy"
    assert report.findings[1].title == "Strict-Transport-Security"
    assert report.target == "https://example.com/"


def test_build_scan_report_handles_missing_tests():
    report = build_scan_report("https://example.com/", {"scan_id": "x"})
    assert report.findings[0].title == "No supported checks were returned"
    assert report.summary.unknown >= 1
